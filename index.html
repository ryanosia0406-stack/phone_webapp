from flask import Flask, request, jsonify, abort from flask_cors import CORS app = Flask(__name__) CORS(app) # enable CORS for all routes @app.route("/location", methods=["POST", "OPTIONS"]) def receive_location(): # ✅ Handle CORS preflight FIRST if request.method == "OPTIONS": return "", 200 # ✅ Only POST reaches here data = request.get_json(silent=True) if not data: abort(400) lat = data.get("latitude") lon = data.get("longitude") print("📍 Received GPS location") print("Latitude:", lat) print("Longitude:", lon) print(str(lat) + " " + str(lon)) print("-" * 30) return jsonify({"status": "ok"}) @app.route("/", methods=["GET"]) def home(): return "GPS Server Running" if __name__ == "__main__": app.run(host="0.0.0.0", port=5000)
